#loocv function
looCV=function(fla,df){
  yhat=numeric(nrow(df))
  
  for (j in 1:nrow(df)){
    train=df[-j,]
    train_lm=lm(fla,data=train)
    yhat[j]=predict(train_lm,newdata=df[j,])
  }
  
  y=df[[all.vars(fla)[1]]]
  sqrt(mean((y-yhat)^2))
}
